2x OLED Ekran (Sol & Sağ Göz) + Arduino Kodu

1 2 Ədəd OLED Ekranın Qoşulması (I2C Ünvanı)

İki ədəd SSD1306 (0.96" və ya 1.3") I2C OLED ekranı Arduino-ya paralel (ikisinin də SDA → A4, SCL → A5, VCC → 5V, GND → GND) qoşmaq üçün ekranların ünvanları fərqli olmalıdır:

  • SOL EKRAN (Sol Göz): Zavod ayarında olduğu kimi qalır → Ünvan: 0x3C
  • SAĞ EKRAN (Sağ Göz): Ekranın arxasındakı 0x78 / 0x7A rezistorunu sağa (0x7A) keçirin → Ünvan: 0x3D

2 Arduino + HC-06 + 2x OLED + Servo Sxemi

HC-06 Bluetooth:

• VCC → 5V

• GND → GND

• TXD → Arduino D2

• RXD → Arduino D3

2x SSD1306 OLED:

• Sol Ekran: 0x3C

• Sağ Ekran: 0x3D

• İkisi də SDA → A4

• İkisi də SCL → A5

SG90 Servo:

• Qırmızı → 5V

• Qəhvəyi → GND

• Narıncı → Arduino D9

3 Arduino Uno / Nano (Sol Ekran = Sol Göz, Sağ Ekran = Sağ Göz) 

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <Servo.h>
#include <SoftwareSerial.h>

// NEXUS-01 Dual OLED Firmware (Sol Ekran = Sol Göz, Sağ Ekran = Sağ Göz)
// HC-06: RX = Pin 2 (HC-06 TXD), TX = Pin 3 (HC-06 RXD)
SoftwareSerial btSerial(2, 3);

// 2 Ayrı OLED Ekran Obyekti (128x64)
Adafruit_SSD1306 leftEyeDisplay(128, 64, &Wire, -1);   // Sol Ekran (Ünvan: 0x3C)
Adafruit_SSD1306 rightEyeDisplay(128, 64, &Wire, -1);  // Sağ Ekran (Ünvan: 0x3D)

Servo headServo;
String state = "IDLE";
int servoPin = 9;
unsigned long lastBlink = 0;

void setup() {
  Serial.begin(9600);
  btSerial.begin(9600);
  
  headServo.attach(servoPin);
  headServo.write(90);
  
  // Sol və Sağ OLED ekranları ayrı-ayrılıqda işə salırıq
  leftEyeDisplay.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  rightEyeDisplay.begin(SSD1306_SWITCHCAPVCC, 0x3D);
  
  drawBothEyes("IDLE", 90, false);
}

void loop() {
  if (btSerial.available()) {
    String cmd = btSerial.readStringUntil('\n');
    processCommand(cmd);
  }
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\n');
    processCommand(cmd);
  }

  // Təbii göz qırpma effekti (hər 3.8 saniyədən bir)
  // Dinləyərkən / düşünərkən / danışarkən daha tez-tez qırpır
  unsigned long blinkEvery = (state == "IDLE") ? 3800 : 1500;
  if (millis() - lastBlink > blinkEvery) {
    lastBlink = millis();
    int ang = headServo.read();
    drawBothEyes(state, ang, true);
    delay(110);
    drawBothEyes(state, ang, false);
  }
}

void processCommand(String cmd) {
  cmd.trim();
  if (cmd.startsWith("S:")) {
    int angle = cmd.substring(2).toInt();
    angle = constrain(angle, 15, 165);
    headServo.write(angle);
    drawBothEyes(state, angle, false);
  } else if (cmd.startsWith("E:")) {
    state = cmd.substring(2);
    drawBothEyes(state, headServo.read(), false);
  }
}

// Hər ekrana öz gözünü çəkən funksiya
void renderSingleEye(Adafruit_SSD1306 &oled, String st, int angle, bool blink, bool isRightEye) {
  oled.clearDisplay();
  
  // Servo bucağına görə göz bəbəyinin sağa/sola baxması
  int offsetX = map(angle, 15, 165, 22, -22);
  int eyeW = 64;
  int eyeH = blink ? 4 : ((st == "SPEAK") ? 46 : (st == "LISTEN" ? 50 : 40));
  
  int x = (128 - eyeW) / 2 + offsetX;
  int y = (64 - eyeH) / 2;

  oled.fillRoundRect(x, y, eyeW, eyeH, blink ? 2 : 12, SSD1306_WHITE);

  if (!blink) {
    // Düşünmə (THINK) rejimində göz qapağı effekti
    if (st == "THINK") {
      oled.fillRect(x - 2, y - 2, eyeW + 4, 14, SSD1306_BLACK);
    }
    // Göz parıltısı (Sol və sağ gözə uyğun simmetrik parlaqlıq)
    int glintX = isRightEye ? (x + 14) : (x + 16);
    oled.fillCircle(glintX, y + 12, 4, SSD1306_BLACK);
  }
  oled.display();
}

void drawBothEyes(String st, int angle, bool blink) {
  renderSingleEye(leftEyeDisplay, st, angle, blink, false);  // Sol ekranda Sol Göz
  renderSingleEye(rightEyeDisplay, st, angle, blink, true);  // Sağ ekranda Sağ Göz
}
